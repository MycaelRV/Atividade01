import java.util.Scanner;

public class TorneioTenis {

    public static void main(String[] args) {
        Scanner ler = new Scanner(System.in);

        int vitorias = 0;

        for (int i = 0; i < 6; i++) {
            String resultado = ler.next();

            if (resultado.equals("V")) {
                vitorias++;
            }
        }

        if (vitorias >= 5) {
            System.out.println(1);
        } else if (vitorias >= 3) {
            System.out.println(2);
        } else if (vitorias >= 1) {
            System.out.println(3);
        } else {
            System.out.println(-1);
        }
    }
}

import java.util.Scanner;

public class CalculoRapido {

    public static void main(String[] args) {
        Scanner ler = new Scanner(System.in);

        int S = ler.nextInt();
        int A = ler.nextInt();
        int B = ler.nextInt();

        int quantidade = 0;

        for (int N = A; N <= B; N++) {

            int numero = N;
            int soma = 0;

            while (numero > 0) {
                soma = soma + numero % 10;
                numero = numero / 10;
            }

            if (soma == S) {
                quantidade++;
            }
        }
    }
}

import java.util.Scanner;

public class Corrida {

    public static void main(String[] args) {
        Scanner ler = new Scanner(System.in);

        int N1 = ler.nextInt();
        int D1 = ler.nextInt();
        int V1 = ler.nextInt();

        int N2 = ler.nextInt();
        int D2 = ler.nextInt();
        int V2 = ler.nextInt();

        double tempo1 = (double) D1 / V1;
        double tempo2 = (double) D2 / V2;

        if (tempo1 < tempo2) {
            System.out.println(N1);
        } else {
            System.out.println(N2);
        }
    }
}
